cmake_minimum_required(VERSION 3.22)
project(lettrine_offline LANGUAGES C CXX)

if(NOT IS_DIRECTORY "${LETTRINE_LLAMA_SOURCE}")
    message(FATAL_ERROR "Set LETTRINE_LLAMA_SOURCE to the prepared llama.cpp source directory.")
endif()

set(BUILD_SHARED_LIBS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GGML_NATIVE OFF)
set(GGML_OPENMP OFF)
set(GGML_LLAMAFILE OFF)
set(GGML_CCACHE OFF)
set(LLAMA_BUILD_COMMON OFF)
set(LLAMA_BUILD_TESTS OFF)
set(LLAMA_BUILD_TOOLS OFF)
set(LLAMA_BUILD_EXAMPLES OFF)
set(LLAMA_BUILD_SERVER OFF)
add_subdirectory("${LETTRINE_LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)

add_library(lettrine_offline SHARED offline_model.cpp)
target_compile_features(lettrine_offline PRIVATE cxx_std_17)
target_compile_options(lettrine_offline PRIVATE -Wall -Wextra)
target_link_options(lettrine_offline PRIVATE "-Wl,--exclude-libs,ALL")
target_link_libraries(lettrine_offline PRIVATE llama)
